## **7. Sieve of Eratosthenes**

The Sieve of Eratosthenes, invented by the Greek mathematician of the same name, provides a quick way to find all prime numbers within a given range. The algorithm involves creating an array of bytes in which positions are “marked” by inserting 1s in the following manner: Beginning with position 2 (which is a prime number), insert a 1 in each array position that is a multiple of 2. Then begin with the next position that is not marked as 1, which is 3, and mark all positions that are multiples of 3. Proceed in the same manner until all multiples of primes have been found. The remaining positions of the array that are unmarked indicate which numbers are prime. For this program, create a 65,000-element array and display all primes between 2 and 65,000. Declare the array in an uninitialized data segment (see Section 3.4.11) and use STOSB to fill it with zeros.

```
정답
.data?
sieve BYTE 65001 DUP(?)

.data
msg BYTE "Prime numbers:",0

.code
main PROC

    ; 배열을 0으로 초기화
    CLD
    MOV EDI, OFFSET sieve
    MOV ECX, 65001
    MOV AL, 0
    REP STOSB

    ; 0과 1은 소수가 아님
    MOV sieve[0], 1
    MOV sieve[1], 1

    MOV EBX, 2

OuterLoop:

    ; EBX * EBX가 65000보다 큰지 확인
    MOV EAX, EBX
    MUL EBX

    CMP EAX, 65000
    JA DisplayPrimes

    ; 이미 표시된 숫자는 건너뜀
    CMP sieve[EBX], 1
    JE NextPrime

    ; 소수의 배수를 표시
    MOV ESI, EAX

MarkLoop:

    CMP ESI, 65000
    JA NextPrime

    MOV sieve[ESI], 1

    ADD ESI, EBX
    JMP MarkLoop

NextPrime:
    INC EBX
    JMP OuterLoop

DisplayPrimes:

    MOV EDX, OFFSET msg
    CALL WriteString
    CALL Crlf

    MOV EBX, 2

PrintLoop:

    CMP EBX, 65000
    JA ProgramDone

    CMP sieve[EBX], 0
    JNE NextNumber

    MOV EAX, EBX
    CALL WriteDec
    CALL Crlf

NextNumber:
    INC EBX
    JMP PrintLoop

ProgramDone:
    EXIT

main ENDP
```

## **8. Bubble Sort**

Add a variable to the BubbleSort procedure in Section 9.5.1 that is set to 1 whenever a pair of values is exchanged within the inner loop. Use this variable to exit the sort before its normal completion if you discover that no exchanges took place during a complete pass through the array. (This variable is commonly known as an exchange flag.)

```
정답
.data
SwapFlag DWORD 0

BubbleSort PROC USES eax ebx ecx edx esi edi,
    pArray:PTR DWORD,
    count:DWORD

    MOV ESI, pArray
    MOV EDI, count

OuterLoop:
    MOV EBX, 0
    MOV ECX, EDI
    DEC ECX

    JZ SortDone

    XOR EDX, EDX

InnerLoop:
    MOV EAX, [ESI + EDX*4]
    MOV ebx, [ESI + EDX*4 + 4]

    CMP EAX, EBX
    JBE SkipSwap

    MOV [ESI + EDX*4], EBX
    MOV [ESI + EDX*4 + 4], EAX

    MOV DWORD PTR [SwapFlag], 1

SkipSwap:
    INC EDX
    LOOP InnerLoop

    CMP SwapFlag, 0
    JE SortDone

    DEC EDI
    MOV SwapFlag, 0
    JMP OuterLoop

SortDone:
    RET

BubbleSort ENDP
```

## **9. Binary Search**

Rewrite the binary search procedure shown in this chapter by using registers for mid, first, and last. Add comments to clarify the registers’ usage.

```
정답
BinarySearch PROC USES ebx ecx edx esi edi,
    pArray:PTR DWORD,
    count:DWORD,
    searchValue:DWORD

    MOV ESI, pArray

    MOV EDI, 0
    ; EDI = first

    MOV EBX, count
    DEC EBX
    ; EBX = last

SearchLoop:

    CMP EDI, EBX
    JG NotFound

    ; mid = (first + last) / 2
    MOV EAX, EDI
    ADD EAX, EBX
    SHR EAX, 1

    MOV ECX, EAX
    ; ECX = mid

    ; 배열[mid]와 검색값 비교
    MOV EDX, [ESI + ECX*4]

    CMP EDX, searchValue
    JE Found

    JL SearchRight

    ; 검색값이 더 작으면 왼쪽 탐색
    LEA EBX, [ECX-1]
    JMP SearchLoop

SearchRight:
    ; 검색값이 더 크면 오른쪽 탐색
    LEA EDI, [ECX+1]
    JMP SearchLoop

Found:
    MOV EAX, ECX
    CLC
    RET

NotFound:
    STC
    RET

BinarySearch ENDP
```

```
    MOV letters[1], AL
    MOV AL, matrix[10]
    MOV letters[2], AL
    MOV AL, matrix[15]
    MOV letters[3], AL

    CALL CheckLetters

    ; 오른쪽 위 -> 왼쪽 아래 대각선
    MOV AL, matrix[3]
    MOV letters[0], AL
    MOV AL, matrix[6]
    MOV letters[1], AL
    MOV AL, matrix[9]
    MOV letters[2], AL
    MOV AL, matrix[12]
    MOV letters[3], AL

    CALL CheckLetters

    EXIT
main ENDP

CheckLetters PROC USES eax ebx ecx esi

    MOV ESI, 0
    MOV EBX, 0

CheckLoop:
    MOV AL, letters[ESI]

    CMP AL, 'A'
    JE CountVowel
    CMP AL, 'E'
    JE CountVowel
    CMP AL, 'I'
    JE CountVowel
    CMP AL, 'O'
    JE CountVowel
    CMP AL, 'U'
    JE CountVowel

    JMP NextLetter

CountVowel:
    INC EBX

NextLetter:
    INC ESI
    CMP ESI, 4
    JL CheckLoop

    CMP EBX, 2
    JNE CheckDone

    MOV ESI, 0

PrintLoop:
    MOV AL, letters[ESI]
    CALL WriteChar
    INC ESI

    CMP ESI, 4
    JL PrintLoop

    CALL Crlf

CheckDone:
    RET

CheckLetters ENDP
END main
```

## **12. Calculating the Sum of an Array Row**

Write a procedure named `calc_row_sum` that calculates the sum of a single row in a two-dimensional array of bytes, words, or doublewords. The procedure should have the following stack parameters: array offset, row size, array type, row index. It must return the sum in EAX. Use explicit stack parameters, not INVOKE or extended PROC. Write a program that tests your procedure with arrays of byte, word, and doubleword. Prompt the user for the row index, and display the sum of the selected row.

```
정답
INCLUDE Irvine32.inc

.data
byteArray BYTE 1,2,3
          BYTE 4,5,6
          BYTE 7,8,9

wordArray WORD 10,20,30
          WORD 40,50,60
          WORD 70,80,90

dwordArray DWORD 100,200,300
            DWORD 400,500,600
            DWORD 700,800,900

prompt BYTE "Enter row index (0-2): ",0
result BYTE "Row sum: ",0

.code
main PROC

    MOV EDX, OFFSET prompt
    CALL WriteString
    CALL ReadInt

    ; 입력한 행 번호를 세 배열에 공통 적용

    PUSH EAX
    PUSH 1
    PUSH 3
    PUSH OFFSET byteArray
    CALL calc_row_sum

    MOV EDX, OFFSET result
    CALL WriteString
    CALL WriteDec
    CALL Crlf

    EXIT
main ENDP

calc_row_sum PROC

    PUSH EBP
    MOV EBP, ESP

    PUSH EBX
    PUSH ECX
    PUSH ESI
    PUSH EDI
    PUSH EDX

    MOV ESI, [EBP+8]     ; 배열 주소
    MOV ECX, [EBP+12]    ; 행의 요소 개수
    MOV EBX, [EBP+16]    ; 자료형 크기
    MOV EDI, [EBP+20]    ; 행 인덱스

    ; 행 시작 주소 계산
    MOV EAX, ECX
    IMUL EAX, EBX
    IMUL EAX, EDI
    ADD ESI, EAX

    XOR EAX, EAX

SumLoop:
    CMP ECX, 0
    JE SumDone

    CMP EBX, 1
    JE ByteSum

    CMP EBX, 2
    JE WordSum

    CMP EBX, 4
    JE DwordSum

    JMP SumDone

ByteSum:
    MOVZX EDX, BYTE PTR [ESI]
    ADD EAX, EDX
    INC ESI
    JMP NextElement

WordSum:
    MOVZX EDX, WORD PTR [ESI]
    ADD EAX, EDX
    ADD ESI, 2
    JMP NextElement

DwordSum:
    ADD EAX, DWORD PTR [ESI]
    ADD ESI, 4

NextElement:
    DEC ECX
    JMP SumLoop

SumDone:
    POP EDX
    POP EDI
    POP ESI
    POP ECX
    POP EBX

    POP EBP
    RET 16

calc_row_sum ENDP
END main
```

## **13. Trimming Leading Characters**

Create a variant of the Str_trim procedure that lets the caller remove all instances of a leading character from a string. For example, if you were to call it with a pointer to the string "###ABC" and pass it the # character, the resulting string would be "ABC."

```
정답
INCLUDE Irvine32.inc

.data
target BYTE "###ABC",0

.code
main PROC

    INVOKE Str_trim, ADDR target, '#'

    MOV EDX, OFFSET target
    CALL WriteString
    CALL Crlf

    EXIT
main ENDP

Str_trim PROC USES eax esi edi,
    pString:PTR BYTE,
    targetChar:BYTE

    MOV ESI, pString

; 제거할 문자를 찾음
SearchLoop:
    MOV AL, [ESI]

    CMP AL, targetChar
    JNE CopyString

    INC ESI
    JMP SearchLoop

; 남은 문자열을 앞으로 복사
CopyString:
    MOV EDI, pString

CopyLoop:
    MOV AL, [ESI]
    MOV [EDI], AL

    CMP AL, 0
    JE TrimDone

    INC ESI
    INC EDI

    JMP CopyLoop

TrimDone:
    RET

Str_trim ENDP
END main
```

## **14. Trimming a Set of Characters**

Create a variant of the Str_trim procedure that lets the caller remove all instances of a set of characters from the end of a string. For example, if you were to call it with a pointer to the string "ABC#$&" and pass it a pointer to an array of filter characters containing "%#!;!$&\*", the resulting string would be "ABC".

```
정답
INCLUDE Irvine32.inc

.data
target BYTE "ABC#$&",0
filter BYTE "%#!;$&*",0

.code
main PROC

    INVOKE Str_trim, ADDR target, ADDR filter

    MOV EDX, OFFSET target
    CALL WriteString
    CALL Crlf

    EXIT
main ENDP

Str_trim PROC USES eax ebx esi edi,
    pString:PTR BYTE,
    pFilter:PTR BYTE

    MOV EBX, pString
    MOV EDI, pString

; 문자열 끝 검색
FindEnd:
    CMP BYTE PTR [EDI], 0
    JE CheckEnd

    INC EDI
    JMP FindEnd

; 마지막 문자 검사
CheckEnd:
    CMP EDI, EBX
    JE TrimDone

    DEC EDI

TrimLoop:
    MOV AL, [EDI]
    MOV ESI, pFilter

FilterLoop:
    MOV DL, [ESI]

    CMP DL, 0
    JE TrimDone

    CMP AL, DL
    JE RemoveChar

    INC ESI
    JMP FilterLoop

; 제거 대상 문자를 널 문자로 변경
RemoveChar:
    MOV BYTE PTR [EDI], 0

    CMP EDI, EBX
    JE TrimDone

    DEC EDI
    JMP TrimLoop

TrimDone:
    RET

Str_trim ENDP
END main
```
